document.getElementById('summarize').addEventListener('click', async () => {
  try {
    statusEl.textContent = 'Preparing...';
    const text = await fetchPageText();
    if (!text) throw new Error('No text found on page');

    const session = await createModelSession();
    statusEl.textContent = 'Summarizing on-device...';
    const simplified = await summarizeText(session, text);

    statusEl.textContent = 'Translating...';
    const targetLang = langSel.value;
    const translated = await translateWithBuiltInAI(simplified, targetLang);

    outputEl.textContent = translated;
    statusEl.textContent = 'Done';
     // After output is produced
    outputEl.textContent = finalText;
    copyBtn.disabled = !outputEl.textContent.trim(); // enable when there’s content
    statusEl.textContent = 'Done';
    setBusy(false); // ensure your setBusy doesn’t override copyBtn when content exists
  } catch (e) {
    statusEl.textContent = `Error: ${e.message}`;
    console.error(e);
  }
 

});
